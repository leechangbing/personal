func binarySearch(nums []int, target int) int {
    left := 0
    right := len(nums)
    for left < right {
        mid := (left + right) / 2
        if nums[mid] == target {
            return mid;
        } else if nums[mid] > target {
            right = mid - 1;
        } else {
            left = mid + 1;
        }
    }
    return -1;
}

// 寻找左侧边界，返回小于目标值的元素个数
func binarySearchLeft(nums []int, target int) int {
    left := 0
    right := len(nums)
    for left < right {
        mid := (left + right) / 2;
        if nums[mid] == target {
            right = mid;
        } else if nums[mid] > target {
            right = mid;
        } else {
            left = mid + 1;
        }
    }
    return left;
}

// 寻找右侧边界，返回小于等于目标值的元素个数
func binarySearchRight(nums []int, target int) int {
    left := 0
    right := len(nums)
    for left < right {
        mid := left + (right-left + 1)/2
        if nums[mid] == target {
            left = mid + 1;
        } else if nums[mid] > target {
            right = mid;
        } else {
            left = mid + 1;
        }
    }
    return left;
}

3. 分治思想
1. 基本思路：
  1. 分割（devide)、求解（conquer)、合并（combine）
  2. 求解某问题的最优解，一般思路是将目标分解为若干个子问题，然后分别求解子问题，最后将子问题最优解组合起来得到目标问题的最优解
2. 实现方式：
  1. 递归：迭代公式，目标问题<->子问题<->基本问题
  2. 递推：记忆化(参数传递或者全局变量、queue/stack)，基本问题->子问题->目标问题
3. 题型：
  1. 暴力算法：遍历
    1. 遍历每一个子问题
    2. 经历每一个决策树分支
  2. 回溯算法(BT)：前进且判断
    1. 在前进的同时进行剪枝操作，不再错误的道路上前进一步
    2. 决策树遍历过程：在遍历过程中判断是否满足约束条件，不满足的时候进行剪枝
  3. 深度优先搜索(DFS)：
    1. 前进方向为深度搜索，每一个方向走到底，然后再更换另一个方向
    2. 适用问题：存在性问题，即仅仅需要找到一个可行解，决策树中的一条可行路径
  4. 广度优先所有(BFS)：
    1. 前进方向为广度搜索，所有方向上同步前进，每次前进一步
    2. 适用问题：最值问题，在所有可行解中找到最合适的解，需要遍历所有可行路径
  5. 动态规划：子问题重复
    1. 如果目标问题分解的子问题之间存在重复，则需要通过额外空间记录答案，避免重复计算提高效率
    2. 实现方式：
      1. 递推：优先考虑此方法，ans(n) = f(ans(n-1), ..., ans(1))，如果ans(n)仅仅和ans(n-1)有关，则可只保留ans(n-1)的状态，没有必要保留ans(1), ..., ans(n-1)的状态；
      2. 递归：需要答案空间ans和标记空间vis，求解子问题查询子问题是否已经解决，如果访问过则直接输出结果，否则需要首次解决；
  6. 贪婪思想：子问题最优解
    1. 目标问题的最优解，由其中一个子问题的最优解得到
    2. 其它算法则是由所有子问题的最优解共同得到，从某种角度来看贪心算法是“稀疏”的
3.1 回溯
1. 基础思路：
  1. 采用暴力遍历的方式通过递归实现，并在遍历的时候判断是否满足约束条件，不在错误的道路上前进一步；
2. 考虑因素：
  1. 路径、选择列表、约束条件
3. 代码框架
  1. 核心：for循环里面的递归，在递归调用之前做选择，在递归之后撤销选择
  2. 复杂度：回溯算法是暴力算法
  3. 总结：维护走过的路径和当前可选列表，并在触发条件结束时，将路径记录到结果集合中
4. 常见问题
  1. 全排列输出：给定n个元素，不存在重复元素，输出所有的排列组合
  2. 全排列输出：给定n个元素，存在重复元素，输出所有的排列组合
  3. 输出1~n中前m个元素的排列
  4. 输出1~n中的所有子集
  5. N皇后问题
5. 算法框架：

3.3 广度优先
1. 一般形式：
  1. 寻找最值解，即最短决策路径
2. 实现方式：
  1. 采用队列(queue)，代价是空间复杂度有点大
3. 双向BFS:
  1. 从起点到终点同时扩散，当两边有交集时停止；
  2. 局限性：必须知道终点在哪里，不再适用队列，而是hash set方便快速判断，两个交集是否具有交集；
4. 算法框架：
type Node struct{
    X int
    Y int
}
func BFS(start Node, target int) int {
    que := list.New()
    vis := set.New()

    que.PushBack(start)
    vis.Insert(start)

    step := 0
    for que.Front() != nil {
        for i:=0; i<len(que); i++{
            curr := que.Front()
            que.PopFront()

            if curr is target{
                return step
            }

            for x:=range(curr) {
                que.PushBack(x)
                vis.Insert(x)
             }
        }
        step++
    }
}
3.4 动态规划
1. 形式：一般是求最值问题，核心问题是穷举方法
2. 特点：存在重叠子问题，通过记忆方式优化穷举过程；具备最优子结构，通过子问题的最值得到原问题的最值；
3. 要素：重叠子问题，最优子结构，状态转移方程
4. 基本步骤：定义状态，定义边界条件，定义状态转移方程
5. 算法框架：
dp := [...][...][...] //定义边界条件
for 状态1 in {状态1|取值}:
    for 状态2 in {状态2|取值}:
        for 状态3 in {状态3|取值}:
            ...//
            dp[状态1][状态2][状态3] = 求最值(选择1, 选择2，选择3)
1. 常见问题：
  1. 斐波那契数列
    1. 递推公式：$$f(n)=f(n-1)+f(n-2)$$
    2. 边界条件：$$f(n) = 1; n=1, 2$$
    3. 合并公式：$$f(n) = \left\{\begin{matrix} f(n-2) + f(n-1), n>2\\ 1, n=1,2 \\ \end{matrix}\right.$$
  2. 维特比译码算法
    1. 递推公式：$$f(n) = min(f(n-1, k) , g(n, k))$$
  3. 凑零钱问题：有n种硬币，面值分别为v1, v2, ..., vn，分别给定非负整数S，选多个硬币是面值之和为S，求最大的数据和最小的数目
    1. 定义：$$ans(S)= \left\{\begin{matrix} -2，无解\\ -1, 未求解\\ 0，S=0\\ >0, 其它\\ \end{matrix}\right.
$$
  4. 完全背包问题：背包容量为S，有n种物品，重量分别为w1, w2, ..., wn，价值分别为v1, v2, ..., vn，在容量限制的情况下，求最大价值
  5. 0-1背包问题：有n种物品，每种只有1个，第i种物品体积为vi，重量为wi，总体积不超过C的条件下，重量尽量大
  6. 无向有环图，最短路径（floyd算法）
    1. 递推公式：$$d(i,j) = min(d(i,k), d(k,j))$$
  7. 最长公共子序列（顺序不变，但不必连续）
    1. 递推公式：
    $$c[i][j]= \left\{\begin{matrix} 0, i=0||j=0\\ c[i-1][j-1], x_i=y_j\\ max(c[i-1][j], c[i][j-1]), x_i!=y_j \end{matrix}\right.$$
8. 最大公共子串(字符位置必须连续）
  1. 递推公式：
  $$c[i][j]= \left\{\begin{matrix} 0, i=0||j=0\\ c[i-1][j-1]+1, x_i = y_j\\ 0, x_i!=y_j \end{matrix}\right.$$

  1. ans[n]表示第n个元素的最大乘积
  2. pos[n]表示第n个元素的最大正乘积
  3. neg[n]表示第n个元素的最大负乘积
10. 最大和区间（股票买卖）：给出股票在n天的价格，分别求在这n天中的买卖股票的最大收益
  1. 递推公式：$$f(n)=max(num(n), g(n-1) + num(n-1))$$
11. 股票问题通解：给定数组，第i个元素表示第i天的价格，可以交易k次，求最大利润
  1. 状态包括三个：天数，最大加一次数，操作（sell, buy, rest)
12. 循环日程表：有n=2^k个人，在n-1天中，两两比赛，每场比赛一天一场，如何安排？
  1. 关键：将2^(k+1)个人分为两组，第一组1~2^k，第二组2^k+1 ~ 2^(k+1)

  1. 关键：令f(n)表示还欠银行的钱，$$f(n)=f(n-1)*(1+x)-c, f(0)=a$$

15. 差分DP
  1. 例题：有N块水晶，任取M块，构建两座塔，使每座塔的高度尽量相等，求最高
  2. 定义状态：d[i][j]表示第i个水晶，高塔和低塔的差为j时高塔的高度
  3. 边界条件：d[0][j]=0, d[i][0]=0

3.5 贪心算法
目标问题的最优解，由其中一个子问题的最优解得到；其它算法则是由全部子问题的最优解得到，从某种角度来看，贪心算法是“稀疏”的；规则来选择具体的子问题，求解之后作为原始问题的答案。
1. 二分查找：目标问题的解要么在左区间，要么在右区间，但事先无法确定，或者无法构造状态转移方程，因此根据判断条件选择具体区间
2. 最优装载问题：有n个物体，重量为w_i，总重量不超过C，最多物品重量
3. 部分背包问题：有n个物品，重量w_i，价值v_i，最大重量为C，使得总价值最大，其中物体可以取走一部分
  1. 目标：“性价比”最高
  2. 策略：首先计算每个物品的价值/重量比例，然后依次取物品
4. 乘船问题：有n个人，重量为w_i，每个船的限重为C，且最多两个人，求最少得船数量
  1. 目标：“浪费”最少
  2. 策略：按照重量排序，双指针方法，i指向最轻的人，j指向最重的人
5. 选择不想交区间：有n个开区间（a_i, b_j)，选择尽量多的区间，使得两两不想交
6. 区间选点问题：有n个比区间[a_i, b_j]，取尽量少的点，使得每个区间内至少有一个点
7. 区间覆盖问题：有n个闭区间[a_i, b_j]，选尽量少的区间覆盖[s, t]
8. Huffman编码：使得平均编码长度最短；
4. 算法专题
4.1 堆(优先队列)
堆是一种完全二叉树，即除了最后一层外，其它所有层都是满的，而最后一层的所有节点都靠左边
最小堆：子节点的数值都小于父节点
1. 定义：如果一棵树的深度为h，除了第h层外，其它各层（1~h-1）的节点上都达到了最大数量，第h层所有的节点都连续集中在最左边，即为完全二叉树；
2. 虽然堆是一种完全二叉树，但并不需要采用树结构保存，对于任意一个父节点的序号n来说，左子节点序号为2n+1，右子节点需要为2n+2，因此可以使用数组保存一个堆；
3. 插入元素：（上浮操作）
  1. 放入尾部
  2. 根据特性，调整位置，保持子节点数值小于父节点位置
func insert(heap []int, x int) []int{
    n := len(heap)
    heap = append(head, x)

    for n > 1 {
        if head[n/2] < heap[n]{
            heap[n], heap[n/2] = heap[n/2], heap[n]
            n /= 2
        } else {
            break
        }
    }
    return heap
}
4. 删除元素：(下沉操作）
  1. 移除堆顶元素
  2. 将堆尾元素放到堆顶
  3. 调整交换位置，保持父节点大于子节点
func remove(heap []int) []int{
    n := len(heap)
    heap[0] = heap[n-1]
    heap = heap[:-1]

    n = n-1
    for n>1 {
        if heap[n/2] < heap[n] {
            heap[n], heap[n/2] = heap[n/2], heap[n]
            n /= 2
        } else {
            break
        }
    }
    return heap
}

3. 移除条件：当前节点不指向任何节点
4. 问题描述：给定一个有向图G[n][n], 若G[u][v]=1则表示节点u指向节点v，若G[u][...]=0则节点u为终点
var G [][]int, vis []int, res[]int
func dfs(int u) bool{
    n := len(G)

    vis[-u] = -1  //正在访问
    for v:=0; v<n; v++{
        if vis[v] == 1 { // 已经访问过
            continue
        } else if vis[v] == -1{ //存在环
            return false
        } else if vis[v] == 0{ // 未访问
            return dfs[v]
        }
    }

    res = append(res, u)
    vis[u] = 1
    return true
}
func topoSort() bool{
    n := len(G)
    for u:=0; u<n; u++{
        if dfs(u)==false{
            return false
        }
    }
    return true
}

// 永不后退匹配
func kmp(next []int, pat string, txt string) int {
    i, j := 0, 0
    m, n := len(pat), len(txt)

    for (i<m) && (j<n) {
        if j==-1|| txt[i] == pat[j] {
            i++
            j++
        } else {
            i = next[i]
        }
    }
    if i==m {
        return j-i
    } else {
        return -1
    }
}
